  function injectCSS() {
    if (document.getElementById('tchiloMeUiV10CSS')) return;
    var st = document.createElement('style');
    st.id = 'tchiloMeUiV10CSS';
    st.textContent = CSS;
    (document.head || document.documentElement).appendChild(st);
  }

  function fixPublishBtn(ed) {
    var btn = ed.querySelector('#mePublish') || ed.querySelector('.me-publish');
    if (btn) {
      btn.textContent = 'PUBLICAR';
      btn.style.cssText = 'border:0;background:transparent;color:#0B0B0C;font:800 17px/1.1 system-ui,-apple-system,sans-serif;letter-spacing:.04em;text-transform:uppercase;padding:8px 4px;margin-left:auto;cursor:pointer;';
    }
    var title = ed.querySelector('#meTitle') || ed.querySelector('.me-top b');
    if (title) title.style.display = 'none';
  }

  function ensureSide(ed) {
    if (ed.querySelector('.me-side-v10')) return;
    var side = document.createElement('div');
    side.className = 'me-side-v10';
    var tools = ed.querySelector('#meTools') || ed.querySelector('.me-tools');
    if (tools) {
      tools.querySelectorAll('button, .me-tool').forEach(function (t) {
        var b = document.createElement('button');
        b.type = 'button';
        b.innerHTML = t.innerHTML;
        b.onclick = function () { t.click(); };
        side.appendChild(b);
      });
    } else {
      ['Filtros', 'Recortar', 'Música', 'Mencionar'].forEach(function (label) {
        var b = document.createElement('button');
        b.type = 'button';
        b.textContent = label;
        b.onclick = function () {
          if (label.indexOf('M') === 0 && label.length > 4) {
            var ms = document.getElementById('meMusicSheet');
            if (ms) ms.classList.add('open');
          }
        };
        side.appendChild(b);
      });
    }
    ed.appendChild(side);
  }

  function ensureMusicSlot(ed) {
    if (ed.querySelector('.me-music-slot-v10')) return ed.querySelector('.me-music-slot-v10');
    var slot = document.createElement('div');
    slot.className = 'me-music-slot-v10';
    slot.innerHTML =
      '<div class="me-music-chip-v10" id="meMusicChipV10"></div>';
    ed.appendChild(slot);
    return slot;
  }

  var _audio = null;
  function stopAudio() {
    if (_audio) {
      try { _audio.pause(); } catch (e) {}
      _audio = null;
